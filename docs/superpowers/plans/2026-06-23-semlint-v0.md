# semlint v0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local-only tool that audits semantic-layer metric definitions and emits a deterministic 0–100 Semantic Trust Score with per-rule findings, remediation, a markdown report, JSON output, and a minimal Streamlit viewer.

**Architecture:** A pure-Python rule engine. `metrics.yml` is parsed into plain dicts; 8 deterministic rule functions each return `Finding` objects; the engine aggregates findings and computes a weighted score; `report.py` serializes to markdown + JSON; `cli.py` and `app.py` are thin shells over the engine. No network, no warehouse, no ML, no randomness — same input always yields the same score.

**Tech Stack:** Python ≥3.11, PyYAML (parsing), Streamlit (viewer only), pytest + ruff (dev). Standard library otherwise.

## Data model (the `metrics.yml` contract)

```yaml
version: 1
dimensions:
  - name: date
    grains: [day, week, month]   # optional; enables ambiguous-grain detection
  - name: region
metrics:
  - name: revenue                # required, the registry key
    label: Total revenue         # optional
    description: ...             # rule 4
    owner: finance@acme.com      # rule 3
    formula: SUM(order_amount)   # rules 2, 8
    grain: day                   # rule 5
    dimensions: [date, region]   # rules 6, 7
    aliases: [total_revenue]     # rule 7
    access: finance              # rule 7
```

## Rule → severity → weight table (locked)

| ID | Rule | Violation severity | Weight |
|----|------|--------------------|--------|
| R1 | duplicate definition (same name ≥2×) | FAIL | 15 |
| R2 | definition collision (same name, ≥2 distinct non-empty formulas) | FAIL | 20 |
| R3 | missing owner | WARN | 8 |
| R4 | missing description | WARN | 7 |
| R5 | undefined grain → FAIL; grain not in `date.grains` → WARN | FAIL/WARN | 15 |
| R6 | orphaned dimension (referenced, not defined) | FAIL | 15 |
| R7 | AI-readiness gaps (missing aliases OR dimensions OR access) | WARN | 12 |
| R8 | non-portable construct (engine-specific SQL token) | WARN | 8 |

Weights sum to 100. `SEVERITY_FACTOR = {FAIL: 1.0, WARN: 0.5, PASS: 0.0}`.

**Score formula (deterministic):** `score = round(100 - Σ_rules[ weight_rule × (Σ_distinct-metrics worst-severity-factor) / metric_count ])`, clamped to `[0, 100]`. A rule with zero violations emits one `PASS` finding and deducts nothing.

```python
from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum


class Severity(str, Enum):
    PASS = "pass"
    WARN = "warn"
    FAIL = "fail"


SEVERITY_FACTOR = {Severity.FAIL: 1.0, Severity.WARN: 0.5, Severity.PASS: 0.0}


@dataclass(frozen=True)
class Finding:
    rule_id: str
    rule_name: str
    severity: Severity
    metric: str | None
    message: str
    remediation: str


@dataclass(frozen=True)
class RuleSpec:
    id: str
    name: str
    weight: int
    violation_severity: Severity


@dataclass(frozen=True)
class AuditResult:
    score: int
    metric_count: int
    rule_count: int
    findings: list[Finding] = field(default_factory=list)
```

```python
from __future__ import annotations

from pathlib import Path

import yaml


def load_metrics(path: str) -> tuple[list[dict], dict[str, dict]]:
    data = yaml.safe_load(Path(path).read_text(encoding="utf-8")) or {}
    metrics = data.get("metrics") or []
    dimensions: dict[str, dict] = {}
    for d in data.get("dimensions") or []:
        if isinstance(d, dict) and d.get("name"):
            dimensions[d["name"]] = d
    return metrics, dimensions
```

```python
from __future__ import annotations

from semlint.contracts import Finding, RuleSpec, Severity

SPEC_DUPLICATE = RuleSpec("R1", "duplicate definition", 15, Severity.FAIL)
SPEC_COLLISION = RuleSpec("R2", "definition collision", 20, Severity.FAIL)


def _pass(spec: RuleSpec) -> Finding:
    return Finding(spec.id, spec.name, Severity.PASS, None, "no issues found", "")


def rule_duplicate_definition(metrics: list[dict], dimensions: dict) -> list[Finding]:
    spec = SPEC_DUPLICATE
    counts: dict[str, int] = {}
    for m in metrics:
        counts[m["name"]] = counts.get(m["name"], 0) + 1
    dupes = sorted(name for name, c in counts.items() if c > 1)
    if not dupes:
        return [_pass(spec)]
    return [
        Finding(
            spec.id,
            spec.name,
            spec.violation_severity,
            name,
            f"metric '{name}' is defined {counts[name]} times",
            f"keep one canonical definition of '{name}' and remove or rename the duplicates",
        )
        for name in dupes
    ]


def rule_definition_collision(metrics: list[dict], dimensions: dict) -> list[Finding]:
    spec = SPEC_COLLISION
    formulas: dict[str, set[str]] = {}
    for m in metrics:
        f = str(m.get("formula") or "").strip()
        formulas.setdefault(m["name"], set())
        if f:
            formulas[m["name"]].add(f)
    colliding = sorted(name for name, fs in formulas.items() if len(fs) > 1)
    if not colliding:
        return [_pass(spec)]
    out = []
    for name in colliding:
        variants = ", ".join(sorted(formulas[name]))
        out.append(
            Finding(
                spec.id,
                spec.name,
                spec.violation_severity,
                name,
                f"metric '{name}' has conflicting formulas: {variants}",
                f"reconcile '{name}' to a single agreed formula",
            )
        )
    return out
```

```python
SPEC_DESCRIPTION = RuleSpec("R4", "missing description", 7, Severity.WARN)


def _missing_field_rule(metrics, spec, field_name, label):
    offenders = sorted(
        (m for m in metrics if not str(m.get(field_name) or "").strip()),

            spec.id,

            spec.violation_severity,
            m["name"],
            f"metric '{m['name']}' has no {label}",
```

```python
SPEC_GRAIN = RuleSpec("R5", "undefined or ambiguous grain", 15, Severity.FAIL)
SPEC_ORPHAN = RuleSpec("R6", "orphaned dimension", 15, Severity.FAIL)


def rule_grain(metrics: list[dict], dimensions: dict) -> list[Finding]:
    spec = SPEC_GRAIN
    allowed = (dimensions.get("date") or {}).get("grains")
    findings = []
    for m in sorted(metrics, key=lambda x: x["name"]):
        grain = str(m.get("grain") or "").strip()
        if not grain:
            findings.append(
                Finding(
                    spec.id,
                    spec.name,
                    Severity.FAIL,
                    m["name"],
                    f"metric '{m['name']}' has no grain",
                    f"declare a time grain for '{m['name']}'",
                )
            )
        elif allowed and grain not in allowed:
            findings.append(
                Finding(
                    spec.id,
                    spec.name,
                    Severity.WARN,
                    m["name"],
                    f"metric '{m['name']}' uses grain '{grain}' not in date.grains {allowed}",
                    f"use one of {allowed} for '{m['name']}' or extend date.grains",
                )
            )
    return findings or [_pass(spec)]


def rule_orphaned_dimension(metrics: list[dict], dimensions: dict) -> list[Finding]:
    spec = SPEC_ORPHAN
    defined = set(dimensions.keys())
    findings = []
    for m in sorted(metrics, key=lambda x: x["name"]):
        for dim in m.get("dimensions") or []:
            if dim not in defined:
                findings.append(
                    Finding(
                        spec.id,
                        spec.name,
                        Severity.FAIL,
                        m["name"],
                        f"metric '{m['name']}' references undefined dimension '{dim}'",
                        f"define dimension '{dim}' or remove it from '{m['name']}'",
                    )
                )
    return findings or [_pass(spec)]
```

```python
    for m in sorted(metrics, key=lambda x: x["name"]):
        missing = []
        if not (m.get("aliases") or []):
            missing.append("aliases/synonyms")
        if not (m.get("dimensions") or []):
            missing.append("allowed dimensions")
        if not str(m.get("access") or "").strip():
            missing.append("access rule")
        if missing:
            joined = ", ".join(missing)
            findings.append(
                Finding(
                    spec.id,
                    spec.name,
                    Severity.WARN,
                    m["name"],
                    f"metric '{m['name']}' missing {joined} for AI grounding",
                    f"add {joined} to '{m['name']}'",
                )
            )
    return findings or [_pass(spec)]


def rule_non_portable(metrics: list[dict], dimensions: dict) -> list[Finding]:
    spec = SPEC_PORTABLE
    findings = []
    for m in sorted(metrics, key=lambda x: x["name"]):
        upper = str(m.get("formula") or "").upper()
        hits = [tok for tok in NON_PORTABLE_TOKENS if tok in upper]
        if hits:
            joined = ", ".join(hits)
            findings.append(
                Finding(
                    spec.id,
                    spec.name,
                    Severity.WARN,
                    m["name"],
                    f"metric '{m['name']}' formula uses non-portable construct(s): {joined}",
                    f"rewrite '{m['name']}' formula using ANSI/OSI-neutral SQL",
                )
            )
    return findings or [_pass(spec)]
```

```python
from semlint.contracts import SEVERITY_FACTOR, AuditResult, Severity
from semlint.rules import ALL_RULES, RULE_WEIGHTS


def run_rules(metrics: list[dict], dimensions: dict) -> list:
    findings = []
    for _spec, fn in ALL_RULES:
        findings.extend(fn(metrics, dimensions))
    return findings


def compute_score(findings: list, metric_count: int) -> int:
    if metric_count <= 0:
        return 0
    worst: dict[str, dict[str, float]] = {}
    for f in findings:
        if f.severity is Severity.PASS:
            continue
        key = f.metric or f"_{f.rule_id}"
        bucket = worst.setdefault(f.rule_id, {})
        factor = SEVERITY_FACTOR[f.severity]
        if factor > bucket.get(key, 0.0):
            bucket[key] = factor
    deduction = 0.0
    for rule_id, per_metric in worst.items():
        weight = RULE_WEIGHTS.get(rule_id, 0)
        deduction += weight * (sum(per_metric.values()) / metric_count)
    return max(0, min(100, round(100 - deduction)))


def run_audit(metrics: list[dict], dimensions: dict) -> AuditResult:
    findings = run_rules(metrics, dimensions)
    score = compute_score(findings, len(metrics))
    return AuditResult(
        score=score,
        metric_count=len(metrics),
        rule_count=len(ALL_RULES),
        findings=findings,
    )


def audit_file(path: str) -> AuditResult:
    metrics, dimensions = load_metrics(path)
    return run_audit(metrics, dimensions)
```

```python
from __future__ import annotations

import json
from pathlib import Path

from semlint.contracts import AuditResult, Severity

_SEV_LABEL = {Severity.PASS: "PASS", Severity.WARN: "WARN", Severity.FAIL: "FAIL"}


def _counts(result: AuditResult) -> dict[str, int]:
    c = {"pass": 0, "warn": 0, "fail": 0}
    for f in result.findings:
        c[f.severity.value] += 1
    return c


def to_markdown(result: AuditResult) -> str:
    c = _counts(result)
    lines = [
        "# semlint audit report",
        "",
        f"**Semantic Trust Score:** {result.score}/100",
        "",
        f"- metrics audited: {result.metric_count}",
        f"- rules run: {result.rule_count}",
        f"- findings: {c['fail']} fail, {c['warn']} warn, {c['pass']} pass",
        "",
        "| rule | severity | metric | finding | remediation |",
        "|---|---|---|---|---|",
    ]
    for f in result.findings:
        metric = f.metric or "—"
        remediation = f.remediation or "—"
        lines.append(
            f"| {f.rule_id} {f.rule_name} | {_SEV_LABEL[f.severity]} | "
            f"{metric} | {f.message} | {remediation} |"
        )
    lines.append("")
    return "\n".join(lines)


def to_json(result: AuditResult) -> str:
    payload = {
        "score": result.score,
        "metric_count": result.metric_count,
        "rule_count": result.rule_count,
        "findings": [
            {
                "rule_id": f.rule_id,
                "rule_name": f.rule_name,
                "severity": f.severity.value,
                "metric": f.metric,
                "message": f.message,
                "remediation": f.remediation,
            }
            for f in result.findings
        ],
    }
    return json.dumps(payload, indent=2, sort_keys=True)


def write_report(result: AuditResult, md_path: str) -> dict[str, str]:
    md = Path(md_path)
    md.parent.mkdir(parents=True, exist_ok=True)
    md.write_text(to_markdown(result), encoding="utf-8")
    json_path = md.with_suffix(".json")
    json_path.write_text(to_json(result), encoding="utf-8")
    return {"markdown": str(md), "json": str(json_path)}
```
